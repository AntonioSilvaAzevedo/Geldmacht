Segmented Control

Controle de abas inline, estilo iOS segmented picker. Alterna sub-telas dentro de um header sem mudar a URL nem re-montar a tela pai. Stateless — a aba ativa fica no componente pai. Import: @/components/ui/segmented-control

Máximo 3 abas. Para 4+ opções use um Select ou a tab bar inferior. O componente é controlled — passe active e onChange; o estado vive na tela pai e a troca de aba não remonta a tela.

Implementação

segmented-control.tsx 
// components/ui/segmented-control.tsx
interface Props {
  tabs:     string[]               // rótulos na ordem · max 3
  active:   string                 // rótulo ativo (controlled)
  onChange: (tab: string) => void
  size?:    'sm' | 'md'            // default 'md'
  fullWidth?: boolean
}

export function SegmentedControl({ tabs, active, onChange, size = 'md', fullWidth }: Props) {
  return (
    <div className="inline-flex gap-1 rounded-[10px] bg-[#2C2C2E] p-[3px]">
      {tabs.map((tab) => (
        <button
          key={tab}
          onClick={() => onChange(tab)}
          className={cn(
            "rounded-lg px-3.5 py-1.5 text-[13px] transition-all",
            tab === active
              ? "bg-black font-semibold text-white shadow-[0_1px_4px_rgba(0,0,0,0.4)]"
              : "text-white/60 hover:text-white",
          )}
        >
          {tab}
        </button>
      ))}
    </div>
  )
}
